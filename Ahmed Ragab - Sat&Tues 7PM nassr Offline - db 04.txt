# Assignment 4 - MySQL

## What you need
- Node.js
- MySQL Server
- A code editor such as VS Code

## How to run the project
1. Install the packages first:
   ```bash
   npm install
   ```
2. Make a copy of `.env.example` and name it `.env`.
3. Open `.env` and write your MySQL password.
4. Run the project:
   ```bash
   npm start
   ```
5. Open `http://localhost:3000`.

When you start the app, it creates the `store_db` database and its three tables if they do not already exist.

## API endpoints

### Products
- `POST /api/products` - create a product
- `GET /api/products` - get all products
- `GET /api/products/:id` - get one product
- `PUT /api/products/:id` - update a product
- `DELETE /api/products/:id` - delete a product
- `GET /api/products/:id/sales` - sales for a product

### Suppliers
- `POST /api/suppliers` - create a supplier
- `GET /api/suppliers` - get all suppliers
- `PUT /api/suppliers/:id` - update a supplier
- `DELETE /api/suppliers/:id` - delete a supplier

### Sales
- `POST /api/sales` - record a sale
- `GET /api/sales` - get all sales

### Database changes
- `POST /api/admin/products/category` - add Category column
- `DELETE /api/admin/products/category` - remove Category column
- `PATCH /api/admin/suppliers/contact-number-type` - change ContactNumber to VARCHAR(15)
- `PATCH /api/admin/products/product-name-not-null` - make ProductName NOT NULL

### Required data and changes
- `POST /api/admin/seed-data` - insert FreshFoods, Milk, Bread, Eggs, and the Milk sale
- `PATCH /api/admin/products/bread-price` - update Bread price to 25.00
- `DELETE /api/admin/products/eggs` - delete Eggs

### Reports
- `GET /api/reports/total-quantity-sold`
- `GET /api/reports/highest-stock`
- `GET /api/reports/suppliers-starting-f`
- `GET /api/reports/products-never-sold`
- `GET /api/reports/sales-details`

## Example request bodies

Create a supplier:
```json
{
  "SupplierName": "FreshFoods",
  "ContactNumber": "01001234567"
}
```

Create a product:
```json
{
  "ProductName": "Milk",
  "Price": 15.00,
  "StockQuantity": 50,
  "SupplierID": 1
}
```

Record a sale:
```json
{
  "ProductID": 1,
  "QuantitySold": 2,
  "SaleDate": "2025-05-20"
}
```

## Other assignment parts
- `sql/musicana_schema.sql` maps the Musicana ERD from Part 1.
- `sql/part2_mapping.sql` maps the User/Product ERD from Part 2.
- `sql/permissions.sql` covers the `store_manager` permissions in Questions 14-16.
- `bonus.txt` contains the LeetCode SQL solution for the bonus question.

## Notes
- Run the required seed endpoint before updating Bread or deleting Eggs.
- The SQL permissions script uses a placeholder password. Change it before running.
- This is for the assignment. Do not make the admin endpoints public without adding authentication first.

require("dotenv").config();

const express = require("express");
const mysql = require("mysql2/promise");

const app = express();
app.use(express.json());

const DB_NAME = process.env.DB_NAME || "store_db";
let pool;

// Create the database, then connect to it.
async function connectDatabase() {
    const connection = await mysql.createConnection({
        host: process.env.DB_HOST || "localhost",
        port: Number(process.env.DB_PORT || 3306),
        user: process.env.DB_USER || "root",
        password: process.env.DB_PASSWORD || ""
    });

    await connection.query(`CREATE DATABASE IF NOT EXISTS \`${DB_NAME}\``);
    await connection.end();

    pool = mysql.createPool({
        host: process.env.DB_HOST || "localhost",
        port: Number(process.env.DB_PORT || 3306),
        user: process.env.DB_USER || "root",
        password: process.env.DB_PASSWORD || "",
        database: DB_NAME,
        waitForConnections: true,
        connectionLimit: 10,
        queueLimit: 0
    });

    await createTables();
    console.log("Database connected and tables are ready.");
}

async function createTables() {
    await pool.query(`
        CREATE TABLE IF NOT EXISTS Suppliers (
            SupplierID INT AUTO_INCREMENT PRIMARY KEY,
            SupplierName VARCHAR(100) NOT NULL,
            ContactNumber VARCHAR(15)
        )
    `);

    await pool.query(`
        CREATE TABLE IF NOT EXISTS Products (
            ProductID INT AUTO_INCREMENT PRIMARY KEY,
            ProductName VARCHAR(100),
            Price DECIMAL(10,2) NOT NULL,
            StockQuantity INT NOT NULL DEFAULT 0,
            SupplierID INT NOT NULL,
            CONSTRAINT fk_products_supplier
                FOREIGN KEY (SupplierID) REFERENCES Suppliers(SupplierID)
        )
    `);

    await pool.query(`
        CREATE TABLE IF NOT EXISTS Sales (
            SaleID INT AUTO_INCREMENT PRIMARY KEY,
            ProductID INT NOT NULL,
            QuantitySold INT NOT NULL,
            SaleDate DATE NOT NULL,
            CONSTRAINT fk_sales_product
                FOREIGN KEY (ProductID) REFERENCES Products(ProductID)
        )
    `);
}

function asyncHandler(fn) {
    return function (req, res, next) {
        Promise.resolve(fn(req, res, next)).catch(next);
    };
}

app.get("/", (req, res) => {

// ==================== PRODUCTS CRUD ====================

// Create a product
app.post("/api/products", asyncHandler(async (req, res) => {
    const { ProductName, Price, StockQuantity, SupplierID } = req.body;

    if (Price === undefined || StockQuantity === undefined || !SupplierID) {
        return res.status(400).json({
            message: "Price, StockQuantity and SupplierID are required."
        });
    }

    const [result] = await pool.query(
        `INSERT INTO Products (ProductName, Price, StockQuantity, SupplierID)
         VALUES (?, ?, ?, ?)`,
        [ProductName ?? null, Price, StockQuantity, SupplierID]
    );

    res.status(201).json({
        message: "Product created successfully.",
        ProductID: result.insertId
    });
}));

// Get all products
app.get("/api/products", asyncHandler(async (req, res) => {
    const [rows] = await pool.query("SELECT * FROM Products");
    res.json(rows);
}));

// Get one product using its ID
app.get("/api/products/:id", asyncHandler(async (req, res) => {
    const [rows] = await pool.query(
        "SELECT * FROM Products WHERE ProductID = ?",

    if (rows.length === 0) {
        return res.status(404).json({ message: "Product not found." });
    }

app.put("/api/products/:id", asyncHandler(async (req, res) => {
    const { ProductName, Price, StockQuantity, SupplierID } = req.body;

    const [result] = await pool.query(
        `UPDATE Products
         SET ProductName = COALESCE(?, ProductName),
             Price = COALESCE(?, Price),
             StockQuantity = COALESCE(?, StockQuantity),
             SupplierID = COALESCE(?, SupplierID)
         WHERE ProductID = ?`,
        [
            ProductName ?? null,
            Price ?? null,
            StockQuantity ?? null,
            SupplierID ?? null,
            req.params.id
        ]
    );

    if (result.affectedRows === 0) {
        return res.status(404).json({ message: "Product not found." });
    }

    res.json({ message: "Product updated successfully." });
}));

app.post("/api/admin/seed-data", asyncHandler(async (req, res) => {
    const connection = await pool.getConnection();

    try {
        await connection.beginTransaction();

        let [suppliers] = await connection.query(
            "SELECT SupplierID FROM Suppliers WHERE SupplierName = ? LIMIT 1",
            ["FreshFoods"]
        );

        let supplierID;
        if (suppliers.length > 0) {
            supplierID = suppliers[0].SupplierID;
        } else {
            const [supplierResult] = await connection.query(
                "INSERT INTO Suppliers (SupplierName, ContactNumber) VALUES (?, ?)",
                ["FreshFoods", "01001234567"]
            );
            supplierID = supplierResult.insertId;
        }

        const products = [
            ["Milk", 15.00, 50],
            ["Bread", 10.00, 30],
            ["Eggs", 20.00, 40]
        ];

        const productIDs = {};

        for (const item of products) {
            const [existing] = await connection.query(
                "SELECT ProductID FROM Products WHERE ProductName = ? LIMIT 1",
                [item[0]]
            );

            if (existing.length > 0) {
                productIDs[item[0]] = existing[0].ProductID;
            } else {
                const [productResult] = await connection.query(
                    `INSERT INTO Products
                     (ProductName, Price, StockQuantity, SupplierID)
                     VALUES (?, ?, ?, ?)`,
                    [item[0], item[1], item[2], supplierID]
                );
                productIDs[item[0]] = productResult.insertId;
            }
        }

        const [sales] = await connection.query(
            "SELECT SaleID FROM Sales WHERE ProductID = ? AND SaleDate = ? AND QuantitySold = ? LIMIT 1",
            [productIDs.Milk, "2025-05-20", 2]
        );

        if (sales.length === 0) {
            await connection.query(
                "INSERT INTO Sales (ProductID, QuantitySold, SaleDate) VALUES (?, ?, ?)",
                [productIDs.Milk, 2, "2025-05-20"]
            );
        }

        await connection.commit();
        res.json({ message: "Required supplier, products and sale are ready." });
    } catch (error) {
        await connection.rollback();
        throw error;
    } finally {
        connection.release();
    }
}));

CREATE DATABASE IF NOT EXISTS Musicana;
USE Musicana;

CREATE TABLE Musician (
    MusicianID INT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    Street VARCHAR(100),
    City VARCHAR(100),
    Phone VARCHAR(20)
);

CREATE TABLE Instrument (
    InstrumentName VARCHAR(100) PRIMARY KEY,
    MusicalKey VARCHAR(30) NOT NULL
);

CREATE TABLE Album (
    AlbumID INT PRIMARY KEY,
    Title VARCHAR(150) UNIQUE NOT NULL,
    CopyrightDate DATE,
    ProducerID INT NOT NULL,
    FOREIGN KEY (ProducerID) REFERENCES Musician(MusicianID)
);

CREATE TABLE Song (
    SongID INT PRIMARY KEY AUTO_INCREMENT,
    Title VARCHAR(150) UNIQUE NOT NULL,
    Author VARCHAR(100) NOT NULL,
    AlbumID INT NOT NULL,
    FOREIGN KEY (AlbumID) REFERENCES Album(AlbumID)
);

-- A musician can play more than one instrument, and an instrument can belong to many musicians.
CREATE TABLE MusicianInstrument (
    MusicianID INT NOT NULL,
    InstrumentName VARCHAR(100) NOT NULL,
    PRIMARY KEY (MusicianID, InstrumentName),
    FOREIGN KEY (MusicianID) REFERENCES Musician(MusicianID),
    FOREIGN KEY (InstrumentName) REFERENCES Instrument(InstrumentName)
);

-- A musician can perform many songs, and a song can be performed by many musicians.
CREATE TABLE MusicianSong (
    MusicianID INT NOT NULL,
    SongID INT NOT NULL,
    PRIMARY KEY (MusicianID, SongID),
    FOREIGN KEY (MusicianID) REFERENCES Musician(MusicianID),
    FOREIGN KEY (SongID) REFERENCES Song(SongID)
);

CREATE DATABASE IF NOT EXISTS Part2Mapping;
USE Part2Mapping;

CREATE TABLE Users (
    id INT PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    firstName VARCHAR(100),
    lastName VARCHAR(100),
    role VARCHAR(50),
    password VARCHAR(255)
);

-- A user can have more than one phone number, so phone numbers go in a separate table.
CREATE TABLE UserPhones (
    userId INT NOT NULL,
    phone VARCHAR(30) NOT NULL,
    PRIMARY KEY (userId, phone),
    FOREIGN KEY (userId) REFERENCES Users(id)
);

CREATE TABLE Products (
    id INT PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    stock INT NOT NULL DEFAULT 0,
    price DECIMAL(10,2) NOT NULL,
    isDeleted BOOLEAN NOT NULL DEFAULT FALSE,
    userId INT NOT NULL,
    FOREIGN KEY (userId) REFERENCES Users(id)
);
